cmake_minimum_required(VERSION 3.25)
project(AIHappeyExplorer LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The Explorer extension requires Windows x64")
endif()
foreach(setting COMMAND_CLSID HOST_NAME APP_NAME)
  if(NOT DEFINED ${setting} OR "${${setting}}" STREQUAL "")
    message(FATAL_ERROR "Missing ${setting}")
  endif()
endforeach()
configure_file(IntegrationConfig.h.in IntegrationConfig.h @ONLY)
add_library(AIHappey.Explorer SHARED ExplorerCommand.cpp ExplorerCommand.def)
target_include_directories(AIHappey.Explorer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(AIHappey.Explorer PRIVATE cxx_std_20)
target_compile_definitions(AIHappey.Explorer PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(AIHappey.Explorer PRIVATE /W4 /WX /EHsc /guard:cf)
target_link_options(AIHappey.Explorer PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
set_property(TARGET AIHappey.Explorer PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(AIHappey.Explorer PRIVATE ole32 shell32 shlwapi advapi32 user32 runtimeobject)
add_executable(ExplorerProbe ExplorerProbe.cpp)
target_compile_features(ExplorerProbe PRIVATE cxx_std_20)
target_compile_definitions(ExplorerProbe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
set_property(TARGET ExplorerProbe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(ExplorerProbe PRIVATE ole32 shell32)
